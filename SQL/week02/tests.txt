select * from employees
select * from departments

select 
	d.department_name 'Department name',max(salary)'Maximum Salary',
	min(salary)'Minimum salary',sum(salary)'Total salary',AVG(salary) 'Average salary',
	count(e.employee_id)'Count employees'

	from
		employees e join departments d
		on d.department_id = e.department_id
		
		group by d.department_name
		having 
			
			count(e.employee_id) >= 5;



select * from employees

select 
	d.department_name'Department name', sum(salary)'Total salary',
	count(e.employee_id)'Number of employee'

	from 
		employees e join departments d
		on e.department_id = d.department_id
	group by
		d.department_name

		sum(salary) > 30000
	order by

select j.job_title,
	d.department_name, count(e.salary)'Number of employee',
	sum(e.salary)'Total salary',avg(e.salary)'Avrege salary'
	from employees e join departments d
	on e.department_id = d.department_id
	join jobs j on e.job_id = j.job_id
	where e.department_id = 80 and
		e.salary >= 7000
	group by j.job_title,department_name
	having sum(e.salary) > 30000
order by [Total salary]desc ,
[Number of employee]desc;
